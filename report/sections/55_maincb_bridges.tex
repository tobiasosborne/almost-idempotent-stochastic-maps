\section{MAIN-CB interface bridges: unit control, witness arithmetic, unit comparison, bijectivity, the merging datum, and inclusion monotonicity}\label{sec:maincb-bridges}

This section records six interface rows used by the MAIN-CB assembly: two scalar and unit
bridges, the compressed-corner unit telescope, the fixed-map bijectivity bridge, the
cross-class four-corner packet itself, and the monotonicity needed when defects are enlarged.
The relevant data are those of Definitions~\ref{def:maincb-witness-ledger},
\ref{def:maincb-partition-state}, and~\ref{def:four-corner-merging-datum}.

\begin{lemma}[\texttt{lem-maincb-isomorphism-unit-control}]\label{lem:maincb-isomorphism-unit-control}
There are universal constants $C_{\mathrm{iso,unit}}<\infty$ and
$e_{\mathrm{iso,unit}}>0$ such that, if $B$ is a finite-dimensional $C^*$-algebra,
$A$ is a finite-dimensional extended $\varepsilon$-$C^*$-algebra, and
$v:B\to A$ is an extended $\delta$-isomorphism in the sense of
Definition~\ref{def:extended-delta-inclusion}, with
$0\le\delta+\varepsilon\le e_{\mathrm{iso,unit}}$, then
\[
  \nrm{v(I_B)-I_A}\le C_{\mathrm{iso,unit}}(\delta+\varepsilon).
\]
The witnesses are independent of dimension, amplification, block data, and the source and target.
\end{lemma}
\noindent\textbf{Registry contract (verbatim).}
\contractquote{There are universal C_iso_unit < infinity and e_iso_unit > 0 such that if B is a finite-dimensional C*-algebra, A is a finite-dimensional extended epsilon-C*-algebra, and v:B->A is an extended delta-isomorphism with 0 <= delta+epsilon <= e_iso_unit, then ||v(I_B)-I_A|| <= C_iso_unit*(delta+epsilon); the witnesses are independent of dimension, amplification, block data, and the particular source and target.}
\paragraph{Proof (prose account of the af-validated tree).}
The tree chooses the numerical witnesses $C_{\mathrm{iso,unit}}=1$ and
$e_{\mathrm{iso,unit}}=1$, before any algebra or map is fixed.
For admissible $B,A,v,\delta,\varepsilon$, the definition of an extended
$\delta$-isomorphism first makes $v$ an extended $\delta$-inclusion.
Specialising its amplification requirement to $n=1$ makes the level-one map
a $\delta$-homomorphism.  The unit clause in that notion then gives directly
$\nrm{v(I_B)-I_A}\le\delta$.  At the same matrix level, the extended
$\varepsilon$-$C^*$ structure and its underlying approximate-algebra axioms give
$|\nrm{I_A}-1|\le\varepsilon$; in particular $\varepsilon\ge0$.
Together with the assumed $\delta\ge0$, this yields
$\delta\le\delta+\varepsilon$.  Hence
$\nrm{v(I_B)-I_A}\le\delta+\varepsilon
=C_{\mathrm{iso,unit}}(\delta+\varepsilon)$.
Because both witnesses were fixed numerical constants, none depends on a matrix
level, dimension, block decomposition, or on the particular $B,A,v$.

\begin{lemma}[\texttt{lem-maincb-compressed-corner-unit-comparison}]\label{lem:maincb-compressed-corner-unit-comparison}
There are universal $C_{\mathrm{corner,unit}}<\infty$ and $e_{\mathrm{corner,unit}}>0$
such that, for every $0\le t\le e_{\mathrm{corner,unit}}$, the following hold at every $n\ge1$.
If $P$ is a $t$-projection in a finite-dimensional extended $t$-$C^*$-algebra $A$, then
\[
 \nrm{I_n\otimes\mathrm{Co}_P(P)-I_n\otimes P}\le C_{\mathrm{corner,unit}}t.
\]
Under the hypotheses of Lemma~\ref{lem:maincb-outer-compression-transfer}, if
$\nrm{v(I_B)-u_{\mathcal S_P}}\le t$ and
$T=\mathrm{Co}^{A_R}_{P^R}\circ\mathrm{Co}^{A}_{R}\circ v$, then
\[
 \nrm{T_n(I_n\otimes I_B)-I_n\otimes P^R}\le C_{\mathrm{corner,unit}}t.
\]
\end{lemma}
\noindent\textbf{Registry contract (verbatim).}
\contractquote{There are universal C_corner_unit < infinity and e_corner_unit > 0 such that both of the following hold: if P is a t-projection in a finite-dimensional extended t-C*-algebra A with 0 <= t <= e_corner_unit, then the compressed-corner unit u_{S_P}=Co_P(P) satisfies ||I_n tensor u_{S_P}-I_n tensor P|| <= C_corner_unit*t for every n >= 1; and, under the hypotheses of lem-maincb-outer-compression-transfer with 0 <= t <= e_corner_unit, if ||v(I_B)-u_{S_P}|| <= t then its explicit outer-compressed map T=Co^{A_R}_{P^R} o Co^A_R o v satisfies ||T_n(I_n tensor I_B)-I_n tensor P^R|| <= C_corner_unit*t for every n >= 1.}
\paragraph{Proof (prose account of the af-validated tree).}
The tree first fixes uniform bounds for a small defective projection $S$: the
$C^*$ lower bound and $\nrm{S^2-S}\le\delta$ give $\nrm S\le2$ after shrinking
the universal threshold, while Definition~\ref{def:compressed-corner} supplies
$\nrm{\mathrm{Co}_{S,S}-L_SR_S}\le K_{\mathrm{cmp}}(\delta+\varepsilon)$.
At level $n$, $S_n=I_n\otimes S$ has the same norm and defect, and amplified
compression identifies $I_n\otimes\mathrm{Co}_S(S)$ with $\mathrm{Co}_{S_n}(S_n)$.
The exact split $S_n(S_n^2)-S_n=S_n(S_n^2-S_n)+(S_n^2-S_n)$ then gives a
universal $K_u(\delta+\varepsilon)$ estimate.  Applied directly to $P_n$, the
live first-clause branch records the explicit coefficient $4K_{\mathrm{cmp}}+7/2$.
For the outer clause, write $a=P^R=\mathrm{Co}^{A}_{R}(P)$ and $A_R=\mathcal S_R^A$.
The corner-algebra and amplified identities make $A_R$ an extended $K_Rt$-$C^*$-algebra
and $a$ a $K_at$-projection; applying the preceding comparison inside $A_R$ yields
$\nrm{I_n\otimes u_{\mathcal S^{A_R}_{P^R}}-I_n\otimes P^R}\le K_{\mathrm{tar}}t$.
The final live branch uses the assumed $\nrm{v(I_B)-u_{\mathcal S_P}}\le t$ and bounded
outer compressions to telescope from $T_n(I_n\otimes I_B)$ to that target unit.
This replacement repaired the challenged unsupported unit-preservation inference;
finite maxima of coefficients and minima of rescaled thresholds give the two witnesses.

\begin{lemma}[\texttt{lem-maincb-cross-datum-bijectivity}]\label{lem:maincb-cross-datum-bijectivity}
Choose $c_0$ for Lemma~\ref{lem:maincb-error-improvement}, its direct-corner-envelope
witnesses, and the witnesses of Lemma~\ref{lem:maincb-compressed-corner-unit-comparison}.
There are universal $C^0_{\mathrm{cross}}\ge1$ and
$0<e^0_{\mathrm{cross}}\le e_{\mathrm{corner,unit}}$, valid also for
Lemma~\ref{lem:maincb-cross-class-merging-datum}, such that the following holds.
Let $W.c_0^{\mathrm{cb}}=c_0$, $W.e_{\mathrm{cross}}\le e^0_{\mathrm{cross}}$, and let
$A,w,U,V,R,v_U,v_V$ satisfy all hypotheses of that merging lemma, including
$R=U\cup V$, the bounds $\varepsilon_U,\varepsilon_V,d_U,d_V\le t\le W.e_{\mathrm{cross}}$,
$d_X\le W.c_0^{\mathrm{cb}}\varepsilon_X$, and
$\nrm{v_X(I_{B_X})-u_{A_X}}\le t$ for $X=U,V$.
Put $B_R=B_U\oplus B_V$, $q_U=(I_{B_U},0)$, $q_V=(0,I_{B_V})$, and
$P_X^R=\mathrm{Co}^{A}_{P_R}(P_X)$.  Define
\begin{align*}
\gamma_{UU}(b_U,0)&=\mathrm{Co}^{A_R}_{P_U^R}(\mathrm{Co}^{A}_{P_R}(v_U(b_U))),&
\gamma_{VV}(0,b_V)&=\mathrm{Co}^{A_R}_{P_V^R}(\mathrm{Co}^{A}_{P_R}(v_V(b_V))),
\end{align*}
and let $\gamma_{UV},\gamma_{VU}$ be the unique maps between the corresponding zero
cross-corners.  These are the four fixed level-one maps of the explicit Stage-3 datum,
and all four maps are bijective.
\end{lemma}

\paragraph{Proof (prose account of the af-validated tree).}
The tree enlarges $c_0$ to at least one and fixes the providers, then sets
$K=\max\{c_0,L,1\}$ and shrinks the merging threshold by

One verifier challenge exposed an omitted premise in the $U$ fixed-map identification;

The four identifications and bijectivity conclusions then assemble at the root.

\begin{lemma}[\texttt{lem-maincb-extended-inclusion-monotone}]\label{lem:maincb-extended-inclusion-monotone}
Let $B$ be a finite-dimensional $C^*$-algebra, let $A$ be a finite-dimensional
extended $\varepsilon$-$C^*$-algebra, let $v:B\to A$ be linear, and suppose
$0\le\delta\le\delta'$.  Every extended $\delta$-inclusion is an extended
$\delta'$-inclusion; every extended $\delta$-isomorphism is an extended
$\delta'$-isomorphism and, in particular, an extended $\delta'$-inclusion.
\end{lemma}
\noindent\textbf{Registry contract (verbatim).}
\contractquote{If B is a finite-dimensional C*-algebra, A is a finite-dimensional extended epsilon-C*-algebra, v:B->A is linear, and 0 <= delta <= delta', then if v is an extended delta-inclusion it is an extended delta'-inclusion, and if v is an extended delta-isomorphism it is an extended delta'-isomorphism and in particular an extended delta'-inclusion.}
\paragraph{Proof (prose account of the af-validated tree).}
Fix an amplification $v_n=I_n\otimes v$ of an extended $\delta$-inclusion.
Definition~\ref{def:extended-delta-inclusion} supplies the two-sided bounds
$(1-\delta)\nrm X\le\nrm{v_n(X)}\le(1+\delta)\nrm X$ and makes $v_n$
a $\delta$-homomorphism.  The byte-matched base definition provisioned after a
verifier challenge spells out bounded linearity, exact involution preservation,
$\nrm{v_n(I)-I}\le\delta$, and
$\nrm{v_n(XY)-v_n(X)v_n(Y)}\le\delta\nrm X\nrm Y$.
Since $0\le\delta\le\delta'$ and all norm factors are nonnegative, the latter
two inequalities remain true with $\delta'$; linearity and involution preservation
do not change.  Likewise
$(1-\delta')\nrm X\le(1-\delta)\nrm X$ and
$(1+\delta)\nrm X\le(1+\delta')\nrm X$.
Thus every amplification is a $\delta'$-homomorphism with the required norm bounds,
so $v$ is an extended $\delta'$-inclusion.  If $v$ was an extended
$\delta$-isomorphism, its bijectivity is independent of the defect parameter;
combining that unchanged bijectivity with the first conclusion proves both remaining claims.
