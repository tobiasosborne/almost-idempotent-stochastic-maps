\section{Convex outsourcing}\label{sec:residual-lower}

\begin{lemma}[Registry contract for \texttt{lem-residual-lower}]
\label{lem:residual-lower}
\contractquote{Convex outsourcing: let C be the convex hull of finitely many points of R^n, and suppose p = sum_{j=1}^m c_j p_j + (1 - s) q with points p_j, q in R^n, coefficients c_j >= 0, s = sum_{j=1}^m c_j < 1, and dist_1(p_j, C) <= dist_1(p, C) for every j; then dist_1(p, C) <= dist_1(q, C).}
\end{lemma}

\noindent\textbf{Status.} \texttt{proved}; \texttt{af: validated}.
This is one of the rigorous in-repo results.

\noindent\textbf{Role in the chain.} This is a frame-free convex
geometry dependency for \texttt{lem-halo-collapse}.  It supplies the
lower residual-distance comparison used when a hidden row is split into
ordinary mass plus a residual term.
