\section{Factorization bound}\label{sec:factorization}

\begin{lemma}[Registry contract for \texttt{lem-factorization}]
\label{lem:factorization}
\contractquote{Factorization bound: let P be an exact signed idempotent (square real matrix with P^2 = P and all row sums equal to 1), let delta(P) = max_i sum_j max(-P_ij, 0), let k = rank(P), and let U = (u_1, ..., u_k) be an actual-row basis of P (the rows p_{u_1}, ..., p_{u_k} of P form a basis of the row space of P) whose Gram volume satisfies Vol(U) >= (1/2) * Vol_max(P), where Vol_max(P) is the maximum Gram volume over all actual-row bases of P; define coordinates a_t(j) by p_j = sum_t a_t(j) p_{u_t}, and for each pivot index s in {1, ..., k} set beta_s(j) = P_{u_s j}, lambda_s(j) = 1 - a_s(j), mu_s(j) = sum_{t != s} max(-a_t(j), 0), sigma_s(j) = sum_{t != s} max(a_t(j), 0), E_s(j) = max(mu_s(j) - lambda_s(j), 0), Phi_s(U) = sum_j max(beta_s(j), 0) * E_s(j), and S*_s(U) = sum_j max(beta_s(j), 0) * (sigma_s(j) + 2 * max(-lambda_s(j), 0)); then for every pivot s, S*_s(U) <= 2 * Phi_s(U) + 6 * delta(P).}
\end{lemma}

\noindent\textbf{Status.} \texttt{proved}; \texttt{af: validated}.
This is one of the rigorous in-repo results.

\noindent\textbf{Proof tree.} The registry shard records an 11-live-node
adversarial tree (12 total nodes, 1 archived), root \texttt{validated},

narrowed to the single inequality before seeding; the inherited
tightness claim for the constants $(2,6)$ was deliberately not elevated
and remains \texttt{proved-mod-audit} material in the registry shard
body.

\noindent\textbf{Role in the chain.} This is the composition link of
the \texttt{(EX)} program: any universal chart bound
$\max_s \Phi_s(U_0) \le C_0\,\delta(P)$ over the $\theta=1/2$ class now
composes rigorously to the class-score constant $C_{\mathrm{sf}} =
2C_0 + 6$.  The \texttt{(EX)} input itself (\ref{conj:ex}) remains a
conjecture: certified numerical evidence supports a plateau near
$C_0 = 2$, but nothing in this section promotes it.
