\section{The H-X financing engine}\label{sec:hx-financing-engine}

The three results of this section form the demand side of the H-X front: an exact
signed idempotent forces two separated points of its row polytope to finance positive
coefficient mass on every high-lever fiber set. All quantities are full-fiber
(clone-invariant) and the picture is signed throughout. For a finite exact signed
idempotent $P$ write $K(P)=\conv\{p_i\}$ for the row polytope, group coordinate
indices into full row-point fibers $Q$ (classes of equal rows), and for a point
$r\in\mathbb R^I$ set $r^+(S)=\sum_{Q\in S}\sum_{j\in Q}\max(r_j,0)$ and
$\nu(r)=\sum_j\max(-r_j,0)$.

\subsection{The basis-free unit transverse moment}

\begin{lemma}[\texttt{lem-hx-transverse-moment-identity}]\label{lem:hx-transverse-moment-identity}
For all points $q_0\ne q_1$ of $K(P)$ and every affine $\chi$ on $\mathbb R^I$ with
$\chi(q_0)=0$ and $\chi(q_1)=1$, the full row-point fibers satisfy
$\sum_Q d_Q\,\chi(p_Q)=1$, where $d_Q=\sum_{j\in Q}(q_{1j}-q_{0j})$.
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{For every finite exact signed idempotent P, all points q0 != q1 of the row polytope K(P) = conv{p_i}, and every affine function chi on R^I with chi(q0) = 0 and chi(q1) = 1, the full row-point fibers satisfy sum_Q d_Q*chi(p_Q) = 1, where d_Q = sum_{j in Q}(q1_j - q0_j).}

\noindent\textbf{Proof sketch.} Every point of $K(P)$ is a convex combination of rows
of $P$ and hence fixed by right multiplication: $q_iP=q_i$. Subtracting,
$D:=q_1-q_0$ is left-fixed, $D=\sum_j D_jp_j$, with $\sum_jD_j=0$ (both points have
coordinate sum one). Applying the linear part $L$ of $\chi$ and using the endpoint
normalization, $1=L(D)=\sum_jD_j\,L(p_j)=\sum_jD_j\,\chi(p_j)$ (the affine constant
dies against $\sum_jD_j=0$); grouping the finite sum by fibers ($\chi(p_Q)$ constant
on a fiber) gives the display. No rank hypothesis, no basis, no coordinate chart.
\hfill$\square$

\subsection{The two-sign-union budget converter}

\begin{lemma}[\texttt{lem-hx-signed-variation-ledger}]\label{lem:hx-signed-variation-ledger}
For every ordered pair $(a,b)$ of points of $K(P)$ and every set $S$ of full
row-point fibers, $\sum_{Q\in S}|d_Q|\le a^+(S)+b^+(S)+\nu(a)+\nu(b)$, where
$d_Q=\sum_{j\in Q}(a_j-b_j)$.
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{For every finite exact signed idempotent P, every ordered pair (a,b) of points of the row polytope K(P), and every set S of full row-point fibers, sum_{Q in S} |d_Q| <= a^+(S) + b^+(S) + nu(a) + nu(b), where d_Q = sum_{j in Q}(a_j - b_j), r^+(S) = sum_{Q in S} sum_{j in Q} max(r_j,0), and nu(r) = sum_j max(-r_j,0).}

\noindent\textbf{Proof sketch.} Partition $S$ by the sign of the \emph{aggregate}
$d_Q$ into $S_+$ and $S_-$ (fibers with $d_Q=0$ contribute nothing). On the genuine
index subset underlying $S_+$,
$\sum_{S_+}d_Q=a(S_+)-b(S_+)\le a^+(S_+)+\nu(b)$, since $a(S_+)\le a^+(S_+)$ and
$-b(S_+)\le\nu(b)$ by the subset ledger of a mass-one point with negative part
$\nu(b)$; symmetrically on $S_-$. Each budget is paid once per sign union, never per
fiber --- the $K$-free discipline of the starvation proof
(\S\ref{sec:starvation-obstruction}), scalarized. \hfill$\square$

\subsection{The financing floor}

\begin{lemma}[\texttt{lem-hx-financing-floor}]\label{lem:hx-financing-floor}
For every ordered pair $(a,b)$ of points of $K(P)$ with $a\ne b$, every affine
$\chi$ with $\chi(a)-\chi(b)=1$, all reals $A>0$ and $\Lambda>0$, and every set $N$
of full row-point fibers with $|\chi(p_Q)|\le A$ on $N$ and $|\chi(p_Q)|\le\Lambda$
off $N$, the complement $F$ of $N$ satisfies
\[
  a^+(F)+b^+(F)\ \ge\ \frac{1-A\,\ell_\chi}{\Lambda}-\nu(a)-\nu(b),
  \qquad \ell_\chi:=\sum_Q|d_Q| .
\]
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{For every finite exact signed idempotent P, every ordered pair (a,b) of points of the row polytope K(P) with a != b, every affine chi with chi(a) - chi(b) = 1, all reals A > 0 and Lambda > 0, and every set N of full row-point fibers with |chi(p_Q)| <= A for every Q in N and |chi(p_Q)| <= Lambda for every Q not in N, the complement F of N satisfies a^+(F) + b^+(F) >= (1 - A*l_chi)/Lambda - nu(a) - nu(b), where l_chi = sum_Q |d_Q| and d_Q = sum_{j in Q}(a_j - b_j).}

\noindent\textbf{Proof sketch.} Recentring $\chi$ at $b$ reduces to the endpoint
normalization of Lemma~\ref{lem:hx-transverse-moment-identity}, whose unit moment
splits over $N\cup F$ as
$1\le A\sum_N|d_Q|+\Lambda\sum_F|d_Q|\le A\,\ell_\chi+
\Lambda\bigl(a^+(F)+b^+(F)+\nu(a)+\nu(b)\bigr)$,
using the lever bounds on each part and
Lemma~\ref{lem:hx-signed-variation-ledger} with $S=F$; rearrange. In the recentred
instantiation ($\chi=\psi_c$ the recentred sign functional of the pair, $N$ a metric
ball around any $c\in K(P)$, $\Lambda=(2+4\delta)/\lone{a-b}$ via the signed
row-diameter bound) the floor reads: \emph{two points at $\ell^1$-separation $\ell$
must jointly ship $\gtrsim(1-A\ell_\chi)\,\ell/(2+4\delta)$ positive mass outside
every low-lever ball} --- the rank-free, slab-free replacement for the W59 lever
geometry. \hfill$\square$

\noindent\textbf{Quantifier correction (W61).} The W60 contract quantified over all
reals $A$; with $N=\varnothing$ that hypothesis is vacuous and the $A<0$ reading is
false (already the $2\times2$ identity idempotent gives a counterexample). The af
elevation surfaced this (challenge \texttt{ch-9388e571}); the contract now states
the $A>0$ form the W60 proof establishes. The retraction of the earlier reading is
