\section{Parameterized transports of the Stage-1 polar layer, II: paths and the inversion derivative}\label{sec:stage1-polar-transports-ii}

The two rows below complete the \texttt{af}-validated transport family begun
in \S\ref{sec:stage1-polar-transports}; the witness datum $W$, its coefficient and margin
fields, the threshold discipline and the \emph{common transport skeleton} (steps
(i)--(v)) are exactly those stated there and are not repeated. What distinguishes these two
rows is that their roots carry \emph{explicit definite descriptions} where their
fixed-constant ancestors bound $u_\delta$, $g_{\varsigma J}$ and $\sigma$ as bare anaphors;
each is therefore built on a spine of \emph{typed} providers that display the very same
$\Pi_\delta$, so that ordinary uniqueness of an inverse --- and nothing weaker --- performs
the identification.

\begin{lemma}[\texttt{lem-stage1-polar-path-transport}]\label{lem:stage1-polar-path-transport}
There exist $C^{0}_{\mathrm{path}},C^{0}_{\mathrm{pol}}\ge1$ and
$\kappa^{0}_{\mathrm{pol}}\in(0,\tfrac12]$ such that for every witness datum $W$ with
$C_{\mathrm{path}}\ge C^{0}_{\mathrm{path}}$, $C_{\mathrm{pol}}\ge C^{0}_{\mathrm{pol}}$
and $0<\kappa_{\mathrm{pol}}\le\kappa^{0}_{\mathrm{pol}}$, for every finite-dimensional
exact-unit $\ecs_r$-$C^*$-algebra, every $\delta>0$ with
$C_{\mathrm{pol}}(\ecs_r+\delta)\le\kappa_{\mathrm{pol}}$, every $U_0,U_1\in\Unit$ and
every $q\in[0,1]$ with $\nrm{U_1-U_0}\le q$, $C_{\mathrm{path}}q\le\tfrac14$ and
$C_{\mathrm{path}}(q+\ecs_rq+q^{2})<\delta-C_{\mathrm{pol}}(\ecs_r\delta+\delta^{2})$:
every $\Lmul{Z_t}$ is invertible and every $Z_t=(1-t)U_0+tU_1$ lies in
$\Unitc{C_{\mathrm{path}}(q+\ecs_rq+q^{2})}$ for $t\in[0,1]$; and, writing $u_\delta$ for
the unique first component of the inverse of $\Pi_\delta$ (the explicit binder of
Lemma~\ref{lem:stage1-polar-retraction-transport}), $H(t,U_0,U_1)=u_\delta(Z_t)$ is jointly
continuous in $(t,U_0,U_1)$, joins $U_0$ to $U_1$, and obeys
$H(t,cU_0,cU_1)=c\,H(t,U_0,U_1)$ for every $c\in U(1)$.
\end{lemma}

\paragraph{Proof (prose account of the af-validated tree).}
\emph{Witnesses first.} Take $(A,B,k)$ from Lemma~\ref{lem:stage1-polar-path-admissibility}
and $(P,\rho)$ from Lemma~\ref{lem:stage1-polar-retraction-transport}, and set
$C^{0}_{\mathrm{path}}=A$, $C^{0}_{\mathrm{pol}}=\max\{B,P\}$,
$\kappa^{0}_{\mathrm{pol}}=\min\{k,\rho\}$ --- all fixed \emph{before} the receiving tuple
$W$ is quantified. With $\gamma=q+\ecs_rq+q^{2}\ge0$ and $y=\ecs_r\delta+\delta^{2}\ge0$,
step (iii) of the skeleton transports each guard: $B(\ecs_r+\delta)\le\kappa_{\mathrm{pol}}\le k$,
$P(\ecs_r+\delta)\le\kappa_{\mathrm{pol}}\le\rho$, $Aq\le C_{\mathrm{path}}q\le\tfrac14$,
and --- the only strict one --- the mixed chain
$A\gamma\le C_{\mathrm{path}}\gamma<\delta-C_{\mathrm{pol}}y\le\delta-By$, whose strictness
is carried by the hypothesised middle inequality alone.

\emph{Binder-free conclusions from the parent.}
Lemma~\ref{lem:stage1-polar-path-admissibility} is used for exactly two things: every
$\Lmul{Z_t}$ is invertible, and $Z_t\in\Unitc{A\gamma}$ --- which radius monotonicity
enlarges to $\Unitc{C_{\mathrm{path}}\gamma}$ with the same right inverse.

\emph{The typed inverse.} Lemma~\ref{lem:stage1-polar-retraction-transport}, applied to the
same tuple $W$ under the non-strict guard $C_{\mathrm{pol}}(\ecs_r+\delta)\le\kappa_{\mathrm{pol}}$,
gives for the \emph{very} map displayed in the root a $C^{1}$ diffeomorphism onto
$S_\delta$ with unique inverse $(u_\delta,h_\delta)$, the normalisations
\eqref{eq:s1pr-inverse}, and $\Unito{\delta-C_{\mathrm{pol}}y}\subseteq S_\delta$. Since
$Z_t\in\Unitc{C_{\mathrm{path}}\gamma}$ gives $\nrm{Z_t\dg\bdot Z_t-J}\le2C_{\mathrm{path}}\gamma
<2(\delta-C_{\mathrm{pol}}y)$, Definition~\ref{def:approximate-unitary-space} puts
$Z_t\in\Unito{\delta-C_{\mathrm{pol}}y}\subseteq S_\delta$; so $u_\delta(Z_t)$ is the
well-typed first component of the inverse of that same displayed $\Pi_\delta$.

\emph{The path.} $(t,U_0,U_1)\mapsto Z_t$ is jointly continuous and $u_\delta$ is $C^{1}$,
hence continuous, on $S_\delta$, so $H$ is jointly continuous; $Z_0=U_0$, $Z_1=U_1$ and
$u_\delta(U)=U$ give the endpoints. For $c\in U(1)$ put $(u,h)=(u_\delta(Z_t),h_\delta(Z_t))$;
conjugate-linearity of $\dagger$, bilinearity of $\bdot$ and $|c|=1$ show directly from
Definition~\ref{def:approximate-unitary-space} that $cu\in\Unit$ (its residual is unchanged
and a right inverse rescales by $\bar c$), while $\Pi_\delta(cu,h)=c\,\Pi_\delta(u,h)=cZ_t$
and $(1-t)cU_0+tcU_1=cZ_t$. A bijection has exactly one inverse, so
$u_\delta(cZ_t)=c\,u_\delta(Z_t)$, i.e.\ $H(t,cU_0,cU_1)=c\,H(t,U_0,U_1)$.

\begin{lemma}[\texttt{lem-stage1-inversion-derivative-transport}]\label{lem:stage1-inversion-derivative-transport}
There exist $C^{0}_{\mathrm{der}},C^{0}_{\mathrm{ch}},C^{0}_{\mathrm{pol}},C^{0}_{\mathrm{grp}}\ge1$
and $\kappa^{0}_{\mathrm{der}},\kappa^{0}_{\mathrm{ch}},\kappa^{0}_{\mathrm{pol}}\in(0,\tfrac12]$
such that for every witness datum $W$ with $C_{\mathrm{der}}\ge C^{0}_{\mathrm{der}}$,
$C_{\mathrm{ch}}\ge C^{0}_{\mathrm{ch}}$, $C_{\mathrm{pol}}\ge C^{0}_{\mathrm{pol}}$,
$C_{\mathrm{grp}}\ge C^{0}_{\mathrm{grp}}$, $0<\kappa_{\mathrm{der}}\le\kappa^{0}_{\mathrm{der}}$,
$0<\kappa_{\mathrm{ch}}\le\kappa^{0}_{\mathrm{ch}}$ and
$0<\kappa_{\mathrm{pol}}\le\kappa^{0}_{\mathrm{pol}}$, for every finite-dimensional
exact-unit $\ecs_r$-$C^*$-algebra, every $\delta>0$, every $\varsigma\in\{+1,-1\}$ and
every $0<r\le\delta$ satisfying the five guards
\eqref{eq:s1id-guards}--\eqref{eq:s1id-retention} at the fields of $W$: writing $u_\delta$
for the unique first component of the inverse of $\Pi_\delta$ (as in
Lemma~\ref{lem:stage1-polar-retraction-transport}) and $g_{\varsigma J}$ for the unique
$C^{1}$ map $B^{\aHerm}_{2\delta}(0)\to B^{\Herm}_{2\delta}(0)$ with
$f_{\varsigma J}(A+g_{\varsigma J}(A))=0$ throughout, define
$\chi_{\varsigma}(A)=\varsigma J\bdot(J+A+g_{\varsigma J}(A))$ and the global $C^{1}$ map
$\sigma(U)=u_\delta(U\dg)$; then $\sigma$ maps $\chi_{\varsigma}(B^{\aHerm}_{r}(0))$ into
the same $\varsigma J$-graph chart, and
$F_{\varsigma}(A)=\phi^{\parallel}_{\varsigma J}(\sigma(\chi_{\varsigma}(A)))$ satisfies
$\nrm{D(F_{\varsigma}-\mathrm{id})(A)+2I_{\aHerm}}\le C_{\mathrm{der}}(\ecs_r+r)$ for every
$A\in B^{\aHerm}_{r}(0)$.
\emph{Transcription note.} The registry writes the sign as $s$; it is renamed $\varsigma$
here exactly as in \S\ref{sec:stage1-smooth-polar}. Where the fixed-constant control row
binds $u_\delta$, $g_{\varsigma J}$ and $\sigma$ as bare anaphors, this row carries
\emph{explicit definite descriptions} for all three --- the point of the transport.
\end{lemma}

\paragraph{Proof (prose account of the af-validated tree).}
\emph{Four fixed providers, then the tuple.} Fix $(G_d,P_d,k_d)$ from
\texttt{lem-stage1-explicit-group-domain-membership}, $(G_c,P_c,k_c)$ from
\texttt{lem-stage1-explicit-group-closeness}, $(P_r,k_r)$ from
Lemma~\ref{lem:stage1-polar-retraction} and $(C_g,k_g)$ from
Lemma~\ref{lem:stage1-unitary-graph-control}, and set $C^{0}_{\mathrm{ch}}=\max\{1,C_g\}$,
$C^{0}_{\mathrm{pol}}=\max\{1,P_d,P_c,P_r\}$, $C^{0}_{\mathrm{grp}}=\max\{1,G_d,G_c\}$,
$\kappa^{0}_{\mathrm{ch}}=\min\{\tfrac12,k_g\}$,
$\kappa^{0}_{\mathrm{pol}}=\min\{\tfrac12,k_d,k_c,k_r\}$ --- all \emph{before} the receiving
tuple enters, so that no step ever absorbs the unbounded receiving $C_{\mathrm{grp}}$. With
$x=\ecs_r\delta+\delta^{2}\ge0$ the receiving guards imply every provider guard by
monotonicity, including $G\ecs_r\le C_{\mathrm{grp}}\ecs_r<\delta-C_{\mathrm{pol}}x\le\delta-Px$
for $(G,P)\in\{(G_d,P_d),(G_c,P_c)\}$ and the chart-retention inequality at $C_g,G_c$. All
polar providers and the root display the identical $\Pi_\delta$ on
$\Unit\times B^{\Herm}_{\delta}(J)$ with target its image, so inverse uniqueness identifies
their inverse pair with the root-bound $(u_\delta,h_\delta)$; uniqueness in
Lemma~\ref{lem:stage1-unitary-graph-control} at $V=\varsigma J$ identifies its graph map
with the root-bound $g_{\varsigma J}$.

\emph{The graph map at $\varsigma J$.} Writing $G=g_{\varsigma J}$ and $Z_T=T+G(T)$, exact
unitality and $(\varsigma J)\dg=\varsigma J$ collapse the graph equation to
$0=G(T)+\tfrac12Z_T\dg\bdot Z_T$; $G=0$ solves it at $T=0$, so $G(0)=0$ by uniqueness, and
the parent's derivative bound $d=C_g(\ecs_r+\delta)\le k_g\le\tfrac12$ gives
$\nrm{G(T)}\le d\nrm{T}$. Differentiating the exact quadratic equation gives
$\nrm{DG(T)}\le(1+\ecs_r)\nrm{Z_T}(1+\nrm{DG(T)})$ with $\nrm{Z_T}\le3r$ for
$\nrm{T}\le2r$, so one universal cutoff yields $\nrm{DG(T)}\le K_gr$ with $K_g=7$.

\emph{The typed factorization.} The root-bound $u_\delta$ is the $C^{1}$ first component
supplied by Lemma~\ref{lem:stage1-polar-retraction}; it is defined at every $U\dg$ by the
explicit domain-membership bridge, and $\dagger$ is real-linear, so
$\sigma(U)=u_\delta(U\dg)$ is globally $C^{1}$ on $\Unit$, with $\sigma(\varsigma J)=\varsigma J$.
For $A\in B^{\aHerm}_{r}(0)$, $U=\chi_{\varsigma}(A)\in\Unit$; put $W=\sigma(U)$ and
$Q=h_\delta(U\dg)-J\in\Herm$ with $\nrm{Q}<\delta$. Then $U\dg=W\bdot(J+Q)$, everything
depending $C^{1}$ on $A$, and the explicit closeness bridge for this same $u_\delta$ gives
$\nrm{W-U\dg}\le G_c\ecs_r$. Exact unitality turns this into $\nrm{W\bdot Q}\le G_c\ecs_r$;
since $W\dg\bdot W=J$, the $C^*$-lower bound and one associator absorb the remainder into
$\nrm{Q}\le K_qG_c\ecs_r$, an estimate in which only the \emph{fixed} $G_c$ occurs.

\emph{Chart retention and the coordinate identity.} From
$U\dg=\varsigma(J-A+G(A))$ one gets $\nrm{U\dg-\varsigma J}\le(1+C_g(\ecs_r+\delta))r$, and
adding the fixed closeness bound the chart-retention guard gives $\nrm{W-\varsigma J}<2\delta$.
As $\Lmul{\varsigma J}=\varsigma I$, $C=\phi_{\varsigma J}(W)$ has norm below $2\delta$; put
$B=C^{\parallel}$. Unitarity of $W$ makes $f_{\varsigma J}(C)=0$, so graph uniqueness forces
$C^{\perp}=G(B)$: $W=\chi_{\varsigma}(B)$ in the \emph{same} chart, with
$B=\phi^{\parallel}_{\varsigma J}(W)=F_{\varsigma}(A)$ and $F_{\varsigma}$ of class $C^{1}$.
Substituting the three factorizations gives the exact identity
$J-A+G(A)=(J+B+G(B))\bdot(J+Q)$, whose anti-Hermitian part is
$-A=B+P^{\parallel}((B+G(B))\bdot Q)$; with $\nrm{Q}\le K_qG_c\ecs_r$ and one universal
cutoff this yields $\nrm{B}\le2r$, so the sharp $\nrm{DG}\le K_gr$ applies at both $A$ and $B$.

\emph{The derivative.} Differentiating that identity along $\xi\in\aHerm$, with
$P=DF_{\varsigma}(A)\xi$ and $R=DQ(A)\xi$, gives $-\xi+DG(A)\xi=P+DG(B)P+R+E$ with
$E=(P+DG(B)P)\bdot Q+(B+G(B))\bdot R$; typing by parts gives $P+\xi=-P^{\parallel}(E)$ and
$R=DG(A)\xi-DG(B)P-P^{\perp}(E)$. The product bound with the three fixed estimates yields
$\nrm{E}\le\alpha\nrm{P}+\beta\nrm{R}$, $\alpha\le K_1\ecs_r$, $\beta\le K_2r$; universal
cutoffs give $\nrm{P}\le2$, then $\nrm{R}\le K_3(\ecs_r+r)$ and
$\nrm{(DF_{\varsigma}(A)+I)\xi}\le D_0(\ecs_r+r)\nrm{\xi}$. Finally, collecting the finitely
many raw cutoffs $c_i$ and all fixed coefficients into $D=\max\{1,D_0,d_1,\dots,d_m\}$ and
setting $C^{0}_{\mathrm{der}}=D$,
$\kappa^{0}_{\mathrm{der}}=\min\{\tfrac12,c_1/2,\dots,c_n/2,1/(2D)\}$ makes the receiving
guards $C_{\mathrm{der}}\ge C^{0}_{\mathrm{der}}$ and
$C_{\mathrm{der}}(\ecs_r+r)\le\kappa_{\mathrm{der}}\le\kappa^{0}_{\mathrm{der}}$ enforce
every cutoff \emph{strictly}, even at the permitted endpoints, and give
$D_0(\ecs_r+r)\le C_{\mathrm{der}}(\ecs_r+r)$. Since
$D(F_{\varsigma}-\mathrm{id})+2I_{\aHerm}=DF_{\varsigma}+I_{\aHerm}$, this is the root
estimate.
