\section{Parameterized transports of the Stage-1 polar layer, II: paths and the inversion derivative}\label{sec:stage1-polar-transports-ii}

\begin{lemma}[\texttt{lem-stage1-polar-path-transport}]\label{lem:stage1-polar-path-transport}
There exist $C^{0}_{\mathrm{path}},C^{0}_{\mathrm{pol}}\ge1$ and
$\kappa^{0}_{\mathrm{pol}}\in(0,\tfrac12]$ such that for every witness datum $W$ with
$C_{\mathrm{path}}\ge C^{0}_{\mathrm{path}}$, $C_{\mathrm{pol}}\ge C^{0}_{\mathrm{pol}}$
and $0<\kappa_{\mathrm{pol}}\le\kappa^{0}_{\mathrm{pol}}$, for every finite-dimensional
exact-unit $\ecs_r$-$C^*$-algebra, every $\delta>0$ with
$C_{\mathrm{pol}}(\ecs_r+\delta)\le\kappa_{\mathrm{pol}}$, every $U_0,U_1\in\Unit$ and
every $q\in[0,1]$ with $\nrm{U_1-U_0}\le q$, $C_{\mathrm{path}}q\le\tfrac14$ and
$C_{\mathrm{path}}(q+\ecs_rq+q^{2})<\delta-C_{\mathrm{pol}}(\ecs_r\delta+\delta^{2})$:
every $\Lmul{Z_t}$ is invertible and every $Z_t=(1-t)U_0+tU_1$ lies in
$\Unitc{C_{\mathrm{path}}(q+\ecs_rq+q^{2})}$ for $t\in[0,1]$; and, writing $u_\delta$ for
the unique first component of the inverse of $\Pi_\delta$ (the explicit binder of
Lemma~\ref{lem:stage1-polar-retraction-transport}), $H(t,U_0,U_1)=u_\delta(Z_t)$ is jointly
continuous in $(t,U_0,U_1)$, joins $U_0$ to $U_1$, and obeys
$H(t,cU_0,cU_1)=c\,H(t,U_0,U_1)$ for every $c\in U(1)$.
\end{lemma}

\paragraph{Proof (prose account of the af-validated tree).}
The skeleton verbatim, seeded by Lemma~\ref{lem:stage1-polar-path-admissibility} with
witnesses $(a,b,k)$ for $(C_{\mathrm{path}},C_{\mathrm{pol}},\kappa_{\mathrm{pol}})$. With
$A=q+\ecs_rq+q^{2}\ge0$ and $B=\ecs_r\delta+\delta^{2}\ge0$, step (iii) transports the
smallness guard and $aq\le C_{\mathrm{path}}q\le\tfrac14$, and the \emph{strict} loss
inequality survives as the chain
$aA\le C_{\mathrm{path}}A<\delta-C_{\mathrm{pol}}B\le\delta-bB$, the strictness carried by

$Z_t\in\Unitc{aA}$; step (v) upgrades the membership to $\Unitc{C_{\mathrm{path}}A}$ by

endpoints, circle equivariance --- already matches verbatim, the bare-$u_\delta$ anaphor of
the parent resolving against the explicit polar binder.

\begin{lemma}[\texttt{lem-stage1-inversion-derivative-transport}]\label{lem:stage1-inversion-derivative-transport}
There exist $C^{0}_{\mathrm{der}},C^{0}_{\mathrm{ch}},C^{0}_{\mathrm{pol}},C^{0}_{\mathrm{grp}}\ge1$
and $\kappa^{0}_{\mathrm{der}},\kappa^{0}_{\mathrm{ch}},\kappa^{0}_{\mathrm{pol}}\in(0,\tfrac12]$
such that for every witness datum $W$ with $C_{\mathrm{der}}\ge C^{0}_{\mathrm{der}}$,
$C_{\mathrm{ch}}\ge C^{0}_{\mathrm{ch}}$, $C_{\mathrm{pol}}\ge C^{0}_{\mathrm{pol}}$,
$C_{\mathrm{grp}}\ge C^{0}_{\mathrm{grp}}$, $0<\kappa_{\mathrm{der}}\le\kappa^{0}_{\mathrm{der}}$,
$0<\kappa_{\mathrm{ch}}\le\kappa^{0}_{\mathrm{ch}}$ and
$0<\kappa_{\mathrm{pol}}\le\kappa^{0}_{\mathrm{pol}}$, for every finite-dimensional
exact-unit $\ecs_r$-$C^*$-algebra, every $\delta>0$, every $\varsigma\in\{+1,-1\}$ and
every $0<r\le\delta$ satisfying the five guards
\eqref{eq:s1id-guards}--\eqref{eq:s1id-retention} at the fields of $W$: writing $u_\delta$
for the unique first component of the inverse of $\Pi_\delta$ (as in
Lemma~\ref{lem:stage1-polar-retraction-transport}) and $g_{\varsigma J}$ for the unique
$C^{1}$ map $B^{\aHerm}_{2\delta}(0)\to B^{\Herm}_{2\delta}(0)$ with
$f_{\varsigma J}(A+g_{\varsigma J}(A))=0$ throughout, define
$\chi_{\varsigma}(A)=\varsigma J\bdot(J+A+g_{\varsigma J}(A))$ and the global $C^{1}$ map
$\sigma(U)=u_\delta(U\dg)$; then $\sigma$ maps $\chi_{\varsigma}(B^{\aHerm}_{r}(0))$ into
the same $\varsigma J$-graph chart, and
$F_{\varsigma}(A)=\phi^{\parallel}_{\varsigma J}(\sigma(\chi_{\varsigma}(A)))$ satisfies
$\nrm{D(F_{\varsigma}-\mathrm{id})(A)+2I_{\aHerm}}\le C_{\mathrm{der}}(\ecs_r+r)$ for every
$A\in B^{\aHerm}_{r}(0)$.
\emph{Transcription note.} The registry writes the sign as $s$; it is renamed $\varsigma$
here exactly as in Lemma~\ref{lem:stage1-inversion-derivative-control}. Where that parent
binds $u_\delta$, $g_{\varsigma J}$ and $\sigma$ as bare anaphors, this row carries
\emph{explicit definite descriptions} for all three --- the point of the transport.
\end{lemma}

\paragraph{Proof (prose account of the af-validated tree).}
The skeleton with \emph{joint} thresholds: each coefficient floor is the maximum, and each
margin cap the minimum, of the corresponding witnesses of the four parents
(Lemmas~\ref{lem:stage1-inversion-derivative-control}, \ref{lem:stage1-polar-retraction},
\ref{lem:stage1-unitary-graph-control} and \ref{lem:stage1-approximate-group-laws}), so
that every field guard transports simultaneously to all four --- including the strict
group-defect chain
$C^{q}_{\mathrm{grp}}\ecs_r\le C_{\mathrm{grp}}\ecs_r<\delta-C_{\mathrm{pol}}y\le\delta-C^{q}_{\mathrm{pol}}y$
(with $y=\ecs_r\delta+\delta^{2}\ge0$) and the radius-chart guard, both monotone in the
same direction. Then three binder identifications discharge the explicit definite
descriptions. \emph{Polar:} Lemma~\ref{lem:stage1-polar-retraction} supplies the bijective
$\Pi_\delta$ on the identical domain with the identical image, and a bijection has exactly
one inverse, so the description selects its first component. \emph{Graph:}
$(\varsigma J)\dg=\varsigma J$ and $(\varsigma J)\bdot(\varsigma J)=J$ exactly, so
$\varsigma J\in\Unit\subseteq\Unitc{\delta}$ and
Lemma~\ref{lem:stage1-unitary-graph-control} at $V=\varsigma J$ supplies the unique
$g_{\varsigma J}$ --- its displayed $f_V$ is verbatim the $f_{\varsigma J}$ above.
\emph{Inversion:} the group laws (Lemma~\ref{lem:stage1-approximate-group-laws}) define
$\sigma(U)=u_\delta(U\dg)$ on all of $\Unit$ for this same $u_\delta$; the graph-control
conclusions satisfy the hypotheses of the smooth-atlas row
(Lemma~\ref{lem:stage1-smooth-unitary-atlas}, via the Neumann invertibility of
$D_{A^{\perp}}f_V$ from \eqref{eq:s1gc-normal}), and, through the \texttt{af}-validated
registry rows \texttt{lem-stage1-smooth-polar-inverse}
(Lemma~\ref{lem:stage1-smooth-polar-inverse}) and
\texttt{lem-stage1-smooth-unitary-operations}
(Lemma~\ref{lem:stage1-smooth-unitary-operations}), the
same $\sigma$ is a smooth --- hence $C^{1}$ --- self-map of $\Unit$, with no point or first
derivative changed. Finally Lemma~\ref{lem:stage1-inversion-derivative-control} applies
with its own witnesses; after the identifications its $\chi_{\varsigma}$, $\sigma$,
$F_{\varsigma}$ and chart are precisely those above, and step (v) enlarges only the
displayed bound, $C^{I}_{\mathrm{der}}(\ecs_r+r)\le C_{\mathrm{der}}(\ecs_r+r)$.
