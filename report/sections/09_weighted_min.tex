\section{Weighted minimum bound}\label{sec:weighted-min}

\begin{lemma}[\texttt{lem-weighted-min}]\label{lem:weighted-min}
Let $p_1,\dots,p_m$ be positive reals with $\sum_i p_i=1$ and let
$n_1,\dots,n_m$ be real numbers. Then
\[
  \min_{i\in\{1,\dots,m\}} n_i\;\le\;\sum_i p_i\,n_i.
\]
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{Weighted minimum bound: let p_1, ..., p_m be positive reals with sum_i p_i = 1 and let n_1, ..., n_m be real numbers; then min over i in {1, ..., m} of n_i <= sum_i p_i * n_i.}

\noindent\textbf{Status.} \texttt{proved}; \texttt{af: validated}.
This is one of the rigorous in-repo results.

\noindent\textbf{Proof tree.} The registry shard records an 8-node
adversarial tree, root \texttt{validated}, taint \texttt{8/8 clean}
(the build run's verifier calls were blanked by a network outage and
the verify phase was resumed cleanly).  The export is
\texttt{proofs/lem-weighted-min/export.md}.

\noindent\textbf{Role in the chain.} Second factored dependency of
\ref{lem:fan-payment}: the index-selection step (some support point
has value at most the weighted average).  Textbook-adjacent, but
registered as its own shard so \texttt{af} trees can import it as a
validated external instead of re-deriving it across siblings.
