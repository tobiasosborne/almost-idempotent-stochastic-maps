\section{Genuine-mass disintegration and top concentration}\label{sec:genuine-disintegration}

Keep the notation of \S\ref{sec:parametric-halo-collapse}: $C=\conv\{p_w:w\in\vis(P)\}$,
$d_j=\distone(p_j,C)$, $\tsq=\sqrt{\negmass}$, height $\height$, and $\posp{P_{ij}}=\max(P_{ij},0)$.
Fix a halo width $a>0$ and the genuine (deep-halo) index set
$G_a=\{j:d_j>a\tsq\}$, assumed nonempty. The two results below are the ``bootstrap'' step: they show
that the mass a row sends into $G_a$ is either supported on genuinely hidden vertices or is small.

\subsection{Disintegration of the halo mass}

\begin{lemma}[\texttt{lem-genuine-disintegration}]\label{lem:genuine-disintegration}
Fix for every row $j$ a convex vertex representation $p_j=\sum_v\lambda_{jv}p_v$ over geometrically
distinct row vertices, and let $g=P\mathbf1_{G_a}$. Then every row index $i$ satisfies
\begin{equation}\label{eq:gd-disintegration}
  g_i\;\le\;M_i^a\;+\;\sum_{j\in G_a}\posp{P_{ij}}\,\frac{\height-d_j}{\height-a\tsq},
  \qquad
  M_i^a=\sum_{j\in G_a}\posp{P_{ij}}\!\!\sum_{v:\,d_v>a\tsq}\!\!\lambda_{jv},
\end{equation}
where $M_i^a\ge0$ is positive mass supported entirely on hidden row vertices of depth in $(a\tsq,\height]$.
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{Genuine-mass disintegration: for an exact signed idempotent P with 0 < delta(P) <= 1/4, nonempty visible set W(P), height H, halo width a > 0 with G_a = {j : dist_1(p_j, conv W) > a*tau} nonempty (tau = sqrt(delta)), fix for every row a vertex representation p_j = sum_v lambda_jv p_v over geometrically distinct row vertices; then every row index i satisfies g_i <= M_i^a + sum_{j in G_a} P_ij^+ * (H - d_j)/(H - a*tau), where g = P*1_{G_a}, d_j = dist_1(p_j, conv W), P_ij^+ = max(P_ij, 0), and M_i^a = sum_{j in G_a} P_ij^+ * sum_{v : d_v > a*tau} lambda_jv is positive mass supported entirely on HIDDEN row vertices at depth in (a*tau, H].}

\noindent\textbf{Proof sketch.} Nonemptiness of $G_a$ forces $\height>a\tsq$. Applying the upper
residual estimate \ref{lem:residual-upper} to the vertex representation gives the depth-convexity bound
$d_j\le\sum_v\lambda_{jv}d_v$. Splitting each $j\in G_a$ into shallow mass
$L_j=\sum_{d_v\le a\tsq}\lambda_{jv}$ and deep mass $U_j=1-L_j$ and using $d_v\le a\tsq$ on the first
class, $d_v\le\height$ on the second, yields $d_j\le a\tsq\,L_j+\height\,U_j$, hence
$L_j\le(\height-d_j)/(\height-a\tsq)$. Since $g_i=\sum_{j\in G_a}P_{ij}\le\sum_{j\in G_a}\posp{P_{ij}}$
and $1=L_j+U_j$, weighting the $L_j$-bound by $\posp{P_{ij}}\ge0$ gives \eqref{eq:gd-disintegration};
the $U_j$-part is $M_i^a$, whose vertices have $d_v>a\tsq$ and hence are hidden of depth
$(a\tsq,\height]$. \hfill$\square$

\subsection{Top concentration}

\begin{lemma}[\texttt{lem-top-concentration}]\label{lem:top-concentration}
Let $v$ be a hidden top vertex of height $\height$ and $a>0$ with $\height>a\tsq$. The positive mass
$v$ places outside the genuine set $G_a$ satisfies
\begin{equation}\label{eq:gd-topconc}
  \sum_{j\notin G_a}\posp{P_{vj}}\;\le\;\frac{\nu_v\,\rowdiam}{\height-a\tsq}.
\end{equation}
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{Top concentration: for an exact signed idempotent P with 0 < delta(P) <= 1/4, nonempty visible set W(P), hidden top vertex v of height H, and halo width a > 0 with H > a*tau (tau = sqrt(delta)), the positive mass v places outside the genuine set satisfies sum_{j notin G_a} P_vj^+ <= nu_v*(2+4*delta)/(H - a*tau), where G_a = {j : dist_1(p_j, conv{p_w : w in W}) > a*tau} and nu_v is the row-v negative mass.}

\noindent\textbf{Proof sketch.} Choose an $\ell_1/\ell_\infty$ support functional $\phi(x)=u\cdot x+b$
with $\linf{u}\le1$, $\phi\le0$ on $C$, and $\phi(p_v)=\height$; set $t_j=\height-\phi(p_j)$. Then
$0\le t_j\le\rowdiam$ (row separation), $t_j\ge\height-d_j\ge\height-a\tsq$ for $j\notin G_a$, and by
row reproduction $\sum_j P_{vj}t_j=0$. The sign split gives
$\sum_j\posp{P_{vj}}t_j=\sum_j\negp{P_{vj}}t_j\le\rowdiam\sum_j\negp{P_{vj}}=\nu_v\rowdiam$. Restricting
to $j\notin G_a$, $(\height-a\tsq)\sum_{j\notin G_a}\posp{P_{vj}}\le\sum_j\posp{P_{vj}}t_j\le\nu_v\rowdiam$;
divide by $\height-a\tsq>0$. \hfill$\square$

\noindent\textbf{Provenance.} \texttt{status: proved}, \texttt{af: validated}; workspace
\texttt{proofs/lem-top-concentration} (export node~1); first-principles support functional, no imports.
The verifier ran an exact LP support-functional test on the banked rank-5 instance (step~4 decider).

\noindent\textbf{Role in the chain.} Lemma~\ref{lem:top-concentration} shows a hidden top vertex sends
all but an $O(\nu_v/\height)$ fraction of its positive mass into the genuine set $G_a$, and
Lemma~\ref{lem:genuine-disintegration} then re-expresses that genuine mass as mass on strictly hidden
vertices plus a controlled shallow remainder. Iterating these against the depth-$d$ collapse
\ref{lem:depth-d-halo-collapse} is the $g$-bootstrap intended to force $\height=O(\tsq)$; the closing
cap remains the open Kernel Conjecture \ref{conj:kernel}.
