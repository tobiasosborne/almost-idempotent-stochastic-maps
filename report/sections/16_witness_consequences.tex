\section{Consequences of the hiddenness witness}\label{sec:witness-consequences}

The scales $\tsq=\sqrt{\negmass}$, $\rho=4\tsq$, $\kappa=\tfrac14\tsq$, the far set $F_v$, and the
exposedness margin $\tstar(v)$ are those of \S\ref{sec:hiddenness-dual-witness}. Write
$C=\conv\{p_w:w\in\vis(P)\}$ and $d_j=\distone(p_j,C)$ for the depth of row $j$; a hidden top vertex $v$
has height $\height=d_v=\max_j d_j$ (\texttt{def-height}). The three results below all price the witness
of Lemma~\ref{lem:hiddenness-dual-witness} against an affine $1$-Lipschitz functional, using the
universal row-separation bound $\lone{p_i-p_k}\le\rowdiam$ of \texttt{def-signed-idempotent}.

\subsection{Depth-Markov concentration}

\begin{lemma}[\texttt{lem-hiddenness-depth-markov}]\label{lem:hiddenness-depth-markov}
Let $v$ be a hidden top vertex of height $\height$ and $(\lambda,\alpha,\beta)$ any hiddenness dual
witness of $v$ with $\sum_i\beta_i<\kappa$. Then for every $c>0$,
\begin{equation}\label{eq:wc-markov}
  \lambda\bigl\{\,f\in F_v : d_f> \height-c\,\tsq\,\bigr\}\;>\;1-\frac{\tfrac12+\negmass}{c}.
\end{equation}
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{Hiddenness depth-Markov: for an exact signed idempotent P with delta(P) > 0, nonempty visible set W(P), hidden top vertex v of height H, and any hiddenness dual witness (lambda, alpha, beta) of v with sum_i beta_i < kappa = tau/4 (tau = sqrt(delta)), one has for every c > 0: lambda{f in F_v : dist_1(p_f, conv{p_w : w in W}) > H - c*tau} > 1 - (1/2 + delta)/c.}

\noindent\textbf{Proof sketch.} Finite-dimensional $\ell_1$ separation at $p_v$ gives an affine
functional $\phi$ with $\phi(p_v)=\height$, linear part of $\ell_\infty$-norm at most $1$, and
$\phi(x)\le\distone(x,C)$; set $\psi=\height-\phi$, so $\psi(p_v)=0$. Then for every row $i$,
\[
  0\le \height-d_i\le\psi(p_i)\le\lone{p_v-p_i}\le\rowdiam .
\]
Applying the linear part of $\psi$ to the witness balance \eqref{eq:hdw-balance} and using
$\psi(p_v)=0$, $\alpha_i,\beta_i\ge0$, and $\sum_i\beta_i<\kappa$ gives
\[
  \sum_{f}\lambda_f\,\psi(p_f)\le\sum_i\beta_i\,\psi(p_i)<\kappa\,\rowdiam=\bigl(\tfrac12+\negmass\bigr)\tsq ,
\]
whence $\sum_f\lambda_f(\height-d_f)<(\tfrac12+\negmass)\tsq$. Markov's inequality for the nonnegative
random variable $\height-d_f$ under the probability $\lambda$ on $F_v$ yields \eqref{eq:wc-markov}.
\hfill$\square$

\noindent\textbf{Provenance.} \texttt{status: proved}, \texttt{af: validated}; workspace
\texttt{proofs/lem-hiddenness-depth-markov} (export node~1), importing
Lemma~\ref{lem:hiddenness-dual-witness}. The hostile verifier \emph{weakened} the hypotheses: the
$c$-parametric bound needs neither tallness nor a $\negmass$ cutoff, only $\sum_i\beta_i<\kappa$, and it
holds for \emph{every} witness tuple.

\subsection{A dual certificate for the margin, and a far top-depth row}

\begin{lemma}[\texttt{lem-row-far-dual-certificate}]\label{lem:row-far-dual-certificate}
Let $v$ be a geometrically distinct row vertex with $\negmass>0$, and set
$L_F(v)=\sum_{f:\,\lone{p_f-p_v}\ge\rho}\posp{P_{vf}}$ with $\posp{P_{vf}}=\max(P_{vf},0)$ and $\nu_v$
the row-$v$ negative mass. If $L_F(v)>0$ then
\begin{equation}\label{eq:wc-cert}
  \tstar(v)\;\le\;\frac{\nu_v}{L_F(v)} .
\end{equation}
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{Row-far dual certificate: for an exact signed idempotent P and a geometrically distinct row vertex v with delta(P) > 0 (tau = sqrt(delta), rho = 4*tau), writing L_F(v) = sum over {f : ||p_f - p_v||_1 >= rho} of max(P_vf, 0) and nu_v the row negative mass, if L_F(v) > 0 then t*(v) <= nu_v / L_F(v), where t*(v) is the exposedness margin of def-exposed.}

\noindent\textbf{Proof sketch.} For any admissible exposer $h$, row reproduction
$p_v=\sum_j P_{vj}p_j$ with $\sum_j P_{vj}=1$ and $h(p_v)=0$ give $0=\sum_j P_{vj}\,h(p_j)$. Splitting
into sign parts and using $0\le h(p_j)\le1$ bounds the positive far contribution:
\[
  \sum_{f\in F_v}\posp{P_{vf}}\,h(p_f)\le\sum_{j:P_{vj}>0}P_{vj}h(p_j)=\sum_{j:P_{vj}<0}(-P_{vj})h(p_j)\le\nu_v .
\]
Dividing the weighted average by $L_F(v)>0$ gives $\min_{f\in F_v}h(p_f)\le\nu_v/L_F(v)$ for every $h$;
taking the supremum over $h$ in \eqref{eq:hdw-tstar} yields \eqref{eq:wc-cert}. \hfill$\square$

\noindent\textbf{Provenance.} \texttt{status: proved}, \texttt{af: validated}; workspace
\texttt{proofs/lem-row-far-dual-certificate} (export node~1); first-principles weak duality (no
attainment needed). Sharp on the banked W29 frontier instance ($\nu_3/L_F=\tfrac1{81}=\tstar(3)$).

\begin{lemma}[\texttt{lem-top-slab-companion}]\label{lem:top-slab-companion}
Let $0<\negmass\le\tfrac12(17-12\sqrt2)$, let $\vis(P)$ be nonempty, and let $v$ be a hidden top vertex
of height $\height>13\,\tsq$. Then there is a row $f$ with $\lone{p_f-p_v}\ge4\tsq$ and
\begin{equation}\label{eq:wc-slab}
  \distone\!\bigl(p_f,\conv\{p_w:w\in\vis\}\bigr)\;>\;\height-\bigl(\tfrac12+\negmass\bigr)\tsq .
\end{equation}
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{Top-slab companion: for an exact signed idempotent P with 0 < delta(P) <= (17 - 12*sqrt(2))/2, nonempty visible set W(P), and hidden top vertex v of height H > 13*tau (tau = sqrt(delta)), there is a row f with ||p_f - p_v||_1 >= 4*tau and dist_1(p_f, conv{p_w : w in W}) > H - (1/2 + delta)*tau.}

\noindent\textbf{Proof sketch.} With the same $\psi=\height-\phi$ as above ($0\le\psi(p_i)\le\rowdiam$),
Lemma~\ref{lem:hiddenness-dual-witness} supplies a witness with
$\sum_f\lambda_f\psi(p_f)\le\sum_i\beta_i\psi(p_i)<\kappa\,\rowdiam=(\tfrac12+\negmass)\tsq$. Since the
$\lambda_f$ are a probability on $F_v$, some $f\in F_v$ has $\psi(p_f)<(\tfrac12+\negmass)\tsq$; that
$f$ satisfies $\lone{p_f-p_v}\ge\rho=4\tsq$ and, via $\phi=\height-\psi\le\distone(\cdot,C)$,
inequality~\eqref{eq:wc-slab}. The constant window $\negmass\le\tfrac12(17-12\sqrt2)$ makes
$13-(\tfrac12+\negmass)\ge4+6\sqrt2>4$, so the far row is genuinely deep. \hfill$\square$

\noindent\textbf{Provenance.} \texttt{status: proved}, \texttt{af: validated}; workspace
\texttt{proofs/lem-top-slab-companion} (export node~1), importing
Lemma~\ref{lem:hiddenness-dual-witness}. The verifier checked the exact symbolic constant chain.

\noindent\textbf{Role in the chain.} Lemma~\ref{lem:hiddenness-depth-markov} shows the witness mass
concentrates within an $O(\tsq)$ slab below the top, Lemma~\ref{lem:row-far-dual-certificate} turns a
positive far mass into a quantitative margin bound, and Lemma~\ref{lem:top-slab-companion} extracts a
single deep far row. Together they are the witness-side inputs to the halo-collapse height cap; the cap
itself (\ref{conj:kernel}) stays open.
