\section{Slab and capacity primitives}\label{sec:slab-capacity-primitives}

The three primitives below are elementary consequences of exact idempotence. For an exact signed
idempotent $P$ (\texttt{def-signed-idempotent}) each row satisfies the \emph{reproduction identity}
\begin{equation}\label{eq:scp-repro}
  p_v=\sum_j P_{vj}\,p_j,\qquad \sum_j P_{vj}=1,
\end{equation}
so for any affine $h$ with $h(p_v)=0$ one has $0=\sum_j P_{vj}\,h(p_j)$. Write
$\posp{P_{vj}}=\max(P_{vj},0)$, $\negp{P_{vj}}=\max(-P_{vj},0)$, and $\nu_v=\sum_j\negp{P_{vj}}$ for the
row-$v$ negative mass (\texttt{def-negative-mass}).

\subsection{The Chebyshev low-slab pincer}

\begin{lemma}[\texttt{lem-cs-low-slab-pincer}]\label{lem:cs-low-slab-pincer}
Let $v$ be any row index, let $h$ be affine with $h(p_v)=0$ and $0\le h(p_j)\le1$ for every row $j$,
and let $s>0$. Then
\begin{equation}\label{eq:scp-pincer}
  \sum_{j:\,h(p_j)\ge s}\posp{P_{vj}}\;\le\;\frac{\nu_v}{s}.
\end{equation}
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{CS low-slab pincer: for an exact signed idempotent P, a row index v with nu_v = sum_j max(-P_vj, 0), an affine h with h(p_v) = 0 and 0 <= h(p_j) <= 1 for every row j, and every s > 0, one has sum over {j : h(p_j) >= s} of max(P_vj, 0) <= nu_v / s.}

\noindent\textbf{Proof sketch.} From \eqref{eq:scp-repro} and $h(p_v)=0$, the sign split gives
$\sum_j\posp{P_{vj}}h(p_j)=\sum_j\negp{P_{vj}}h(p_j)\le\sum_j\negp{P_{vj}}=\nu_v$, using
$0\le h(p_j)\le1$. On the slab $A_s=\{j:h(p_j)\ge s\}$, $s\,\posp{P_{vj}}\le\posp{P_{vj}}h(p_j)$, so
$s\sum_{A_s}\posp{P_{vj}}\le\sum_j\posp{P_{vj}}h(p_j)\le\nu_v$; divide by $s>0$. \hfill$\square$

\noindent\textbf{Provenance.} \texttt{status: proved}, \texttt{af: validated}; workspace
\texttt{proofs/lem-cs-low-slab-pincer} (export node~1). The verifier confirmed the minimal hypotheses
(no hiddenness, no vertex, no $\negmass>0$) and that equality is attained at $s=\tstar$ on the W29
frontier instance, so \eqref{eq:scp-pincer} is sharp.

\subsection{Row-zero capacity}

\begin{lemma}[\texttt{lem-row-zero-capacity}]\label{lem:row-zero-capacity}
Let $i$ be a row index and $y$ a vector with $Py=y$, $0\le y_j\le1$ for all $j$, $y_i=0$, and
$y_f\ge\kappa$ for all $f$ in a set $F$. Then
\begin{equation}\label{eq:scp-capacity}
  \kappa\sum_{f\in F}\posp{P_{if}}\;\le\;\nu_i .
\end{equation}
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{Row-zero capacity: for an exact signed idempotent P, a row index i, and any vector y with P y = y, 0 <= y_j <= 1 for all j, y_i = 0, and y_f >= kappa for all f in a set F, one has kappa * sum over f in F of max(P_if, 0) <= nu_i, where nu_i is the row-i negative mass.}

\noindent\textbf{Proof sketch.} The $i$-th coordinate of $Py=y$ with $y_i=0$ gives
$0=\sum_j P_{ij}y_j$, hence $\sum_j\posp{P_{ij}}y_j=\sum_j\negp{P_{ij}}y_j\le\sum_j\negp{P_{ij}}=\nu_i$
by $0\le y_j\le1$. On $F$, $y_f\ge\kappa$ gives
$\kappa\sum_{f\in F}\posp{P_{if}}\le\sum_{f\in F}\posp{P_{if}}y_f\le\sum_j\posp{P_{ij}}y_j\le\nu_i$.
\hfill$\square$

\noindent\textbf{Provenance.} \texttt{status: proved}, \texttt{af: validated}; workspace
\texttt{proofs/lem-row-zero-capacity} (export node~1), resting on
Lemma~\ref{lem:harmonic-affine-bridge} in the registry (which certifies the $Py=y$ hypothesis is the
harmonic condition). The verifier made the $y_i=0$ hypothesis explicit.

\subsection{The harmonic--affine bridge}

\begin{lemma}[\texttt{lem-harmonic-affine-bridge}]\label{lem:harmonic-affine-bridge}
For an exact signed idempotent $P$ with rows $p_i=(P_{ij})_j$, a vector $g$ satisfies $Pg=g$ if and
only if there is a vector $u$ with $g_i=u\cdot p_i$ for every row index $i$.
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{Harmonic-affine bridge: for an exact signed idempotent P with rows p_i = (P_ij)_j, a vector g satisfies Pg = g if and only if there exists u with g_i = u . p_i for every row index i; in the forward direction u = g works (g_i = p_i . g), and the constant term of any affine representation is absorbable into u since all row sums equal 1.}

\noindent\textbf{Proof sketch.} $(\Rightarrow)$ If $Pg=g$ then $g_i=(Pg)_i=p_i\cdot g$, so $u=g$ works.
$(\Leftarrow)$ If $g_i=u\cdot p_i$ then by row reproduction \eqref{eq:scp-repro},
$(Pg)_i=\sum_j P_{ij}(u\cdot p_j)=u\cdot\bigl(\sum_j P_{ij}p_j\bigr)=u\cdot p_i=g_i$. A constant term
$c$ in an affine representation $g_i=c+u\cdot p_i$ is absorbed by $u'=u+c\mathbf1$, since every row sum
is $1$. \hfill$\square$

\noindent\textbf{Provenance.} \texttt{status: proved}, \texttt{af: validated}; workspace
\texttt{proofs/lem-harmonic-affine-bridge} (export node~1); two lines each direction, checked
entrywise on the banked W19 rank-5 fixture.

\noindent\textbf{Role in the chain.} Lemma~\ref{lem:harmonic-affine-bridge} identifies the $P$-harmonic
vectors ($Pg=g$) with affine test functions of the rows, so an admissible exposer is exactly a
box-constrained harmonic vector; Lemmas~\ref{lem:cs-low-slab-pincer}
and~\ref{lem:row-zero-capacity} are then the two capacity estimates that price positive far mass against
$\nu_v$. They feed the same halo-collapse and starvation arguments as the witness lemmas of
\S\ref{sec:witness-consequences}.
