\section{The closed polar retraction and its coherence}\label{sec:stage1-polar-retraction}

\begin{lemma}[\texttt{lem-stage1-polar-retraction}]\label{lem:stage1-polar-retraction}
There are universal $C_{\mathrm{pol}}\ge1$ and $\kappa_{\mathrm{pol}}\in(0,\tfrac12]$ such
that for every finite-dimensional exact-unit $\ecs_r$-$C^*$-algebra and every $\delta>0$
with $C_{\mathrm{pol}}(\ecs_r+\delta)\le\kappa_{\mathrm{pol}}$, the map
\begin{equation}\label{eq:s1pr-pi}
  \Pi_\delta(U,H)=U\bdot H
\end{equation}
is a $C^{1}$ diffeomorphism from $\Unit\times B^{\Herm}_{\delta}(J)$ onto an open set
$S_\delta$; its inverse $(u_\delta,h_\delta)$ obeys
\begin{equation}\label{eq:s1pr-inverse}
  X=u_\delta(X)\bdot h_\delta(X),\qquad u_\delta(U)=U,\qquad h_\delta(U)=J\quad(U\in\Unit);
\end{equation}
and, writing $D=\ecs_r\delta+\delta^{2}$,
\begin{equation}\label{eq:s1pr-sandwich}
  \Unito{\delta-C_{\mathrm{pol}}D}\;\subseteq\;S_\delta\;\subseteq\;\Unito{\delta+C_{\mathrm{pol}}D}.
\end{equation}
\emph{Transcription note.} Only the \emph{shrunken} inner set of \eqref{eq:s1pr-sandwich} is
claimed; no larger inner radius is asserted. Both $\Unito{\cdot}$ are the strict-inequality
sets of Definition~\ref{def:approximate-unitary-space}, so membership always carries the
right-inverse clause as well as the defect bound.
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{Closed C^1 polar retraction: there are universal C_pol >= 1, kappa_pol in (0, 1/2] such that for every finite-dimensional exact-unit epsilon_r-C*-algebra and delta > 0 with C_pol*(epsilon_r + delta) <= kappa_pol, Pi_delta(U, H) = U bold-dot H is a C^1 diffeomorphism from calU x B^{calH}_delta(J) onto an open S_delta, its inverse (u_delta, h_delta) obeys X = u_delta(X) bold-dot h_delta(X), u_delta(U) = U, h_delta(U) = J, and calU_{delta - C_pol*(epsilon_r*delta + delta^2)} subseteq S_delta subseteq calU_{delta + C_pol*(epsilon_r*delta + delta^2)}.}

\paragraph{Proof (prose account of the af-validated tree).}
The constants $C_{\mathrm{pol}},\kappa_{\mathrm{pol}}$ are fixed at the end as a finite
maximum and a finite positive minimum of the thresholds used below, so they remain
universal.

\emph{The coordinate map.} Fix a base $V\in\Unitc{r}$ with $r\le c\delta$ ($c=4$ is the only
scale used). As in \S\ref{sec:stage1-polar-charts}, $\nrm{\Lmul{V\dg}\Lmul{V}-I}\le
\ecs_r\nrm{V}^{2}+2(1+\ecs_r)r$, which the guard makes at most $\tfrac14$; hence $\Lmul{V}$
is injective, therefore bijective in finite dimension, with a universal bound on
$\nrm{\Lmul{V}^{-1}}$. Writing $A(a)=a+g_V(a)$ for the graph map of
Lemma~\ref{lem:stage1-unitary-graph-control} at scale $c\delta$, exact unitality and
bilinearity give the \emph{exact} identity
\begin{equation}\label{eq:s1pr-coord}
  p_V(a,q):=\Lmul{V}^{-1}\bigl(\Pi_\delta(V\bdot(J+A(a)),\,J+q)-V\bigr)
           =A(a)+q+\Lmul{V}^{-1}\bigl((V\bdot A(a))\bdot q\bigr),
\end{equation}
with no reassociation. Differentiating \eqref{eq:s1pr-coord} and using
$\nrm{Dg_V}\le C_{\mathrm{ch}}(\ecs_r+c\delta)$, $\nrm{A(a)}\le K\delta$ and the
product-norm axiom bounds
\begin{equation}\label{eq:s1pr-nearid}
  \sup\nrm{J_0^{-1}\bigl(Dp_V-J_0\bigr)}\;\le\;C_0(\ecs_r+\delta)\;=:\;\theta\;\le\;\tfrac14
\end{equation}
on the convex box $\nrm{a}<2c\delta$, $\nrm{q}<\tfrac32\delta$, where
$J_0(\alpha,\beta)=\alpha+\beta$ is the real-linear isomorphism
$\aHerm\oplus\Herm\to\calX$ (of inverse norm at most $1$, because the involution is
isometric). At the base point, $p_V(0,0)=g_V(0)$, so \eqref{eq:s1gc-bounds} gives
$\nrm{p_V(0,0)+\tfrac12(V\dg\bdot V-J)}\le C_0 D$.

\emph{Existence of the polar factorisation.} For $V\in\Unito{\delta-C_{\mathrm{pol}}D}$ the
base-point estimate reads $\nrm{p_V(0,0)}<\delta-(C_{\mathrm{pol}}-K_b)D$; choosing
$C_{\mathrm{pol}}>K_b+K_d$ and shrinking the guard so that $\theta<\tfrac14$ makes this at
most $(1-\theta)\delta$, so some $\rho<\delta$ satisfies
$\nrm{p_V(0,0)}/(1-\theta)<\rho$. On the closed $\rho$-ball in the max norm,
$F(z)=z-p_V(z)$ is a $\theta$-contraction into itself; its unique fixed point $z=(a,q)$
solves $p_V(a,q)=0$, i.e.
\begin{equation}\label{eq:s1pr-factor}
  V=\bigl[V\bdot(J+a+g_V(a))\bigr]\bdot(J+q),\qquad \nrm{q}<\delta,
\end{equation}
with the bracket in $\Unit$ by Lemma~\ref{lem:stage1-unitary-graph-control}. Integrating
\eqref{eq:s1pr-nearid} along segments makes $p_V$ injective on the whole box, so this zero
is the only one there.

\emph{Global injectivity and the diffeomorphism.} Suppose $X=U_i\bdot(J+q_i)$ with
$U_i\in\Unit$ and $\nrm{q_i}<\delta$ for $i=1,2$. Expanding
$X\dg\bdot X-J$ around $q=0$ without losing the $q$ factor shows $X\in\Unitc{4\delta}$ and
$\Lmul{X}$ invertible, so one may take the graph chart \emph{at $X$ itself} at scale
$4\delta$: with $A_i=\Lmul{X}^{-1}(U_i-X)$, the vanishing of $f_X(A_i)$ and uniqueness force
$A_i^{\perp}=g_X(A_i^{\parallel})$, and the near-identity estimate for $p_X$ on that box
gives $(a_1,q_1)=(a_2,q_2)$, hence $U_1=U_2$ and $q_1=q_2$. Thus $\Pi_\delta$ is injective;
by \eqref{eq:s1pr-nearid} its derivative is invertible at every point, so the
finite-dimensional $C^{1}$ inverse function theorem makes it a local $C^{1}$ diffeomorphism,
$S_\delta$ open, and the local inverses glue to a $C^{1}$ inverse $(u_\delta,h_\delta)$ with
$X=u_\delta(X)\bdot h_\delta(X)$. Exact unitality gives $U=U\bdot J$, whence
$u_\delta(U)=U$ and $h_\delta(U)=J$.

\emph{The radius sandwich.} For $X=U\bdot(J+q)\in S_\delta$, the star rule expands
$X\dg\bdot X-J$ into two cross terms --- each equal to $q$ up to an associator of norm
$O(\ecs_r\delta)$ --- plus one quadratic term of norm $O(\delta^{2})$, giving
$\nrm{X\dg\bdot X-J}<2\delta+C_{\mathrm{out}}D$; and $\Lmul{X}$ is a small perturbation of
the invertible $\Lmul{U}$, so $X$ has a right inverse. Taking
$C_{\mathrm{pol}}\ge C_{\mathrm{out}}/2$ gives the outer inclusion of
\eqref{eq:s1pr-sandwich}. The inner inclusion is \eqref{eq:s1pr-factor}.

\begin{lemma}[\texttt{lem-stage1-polar-coherence-naturality}]\label{lem:stage1-polar-coherence-naturality}
For every exact-unit algebra and every two polar data $(\delta_j,S_j,u_j,h_j)$, $j=1,2$, for
which $\Pi_{\delta_j}:\Unit\times B^{\Herm}_{\delta_j}(J)\to S_j$,
$(U,H)\mapsto U\bdot H$, is bijective with inverse $(u_j,h_j)$, one has
\begin{equation}\label{eq:s1cn-coherence}
  (u_1,h_1)=(u_2,h_2)\qquad\text{on } S_1\cap S_2 ;
\end{equation}
moreover, for $c\in U(1)$ and $X,cX\in S_j$,
\begin{equation}\label{eq:s1cn-natural}
  u_j(cX)=c\,u_j(X),\qquad h_j(cX)=h_j(X).
\end{equation}
\emph{Transcription note.} This row is \emph{conditional}: the polar data are hypotheses and
no free polar witness is produced. It is never used as an existence assertion --- existence
is Lemma~\ref{lem:stage1-polar-retraction} --- but only to identify inverse components that
have already been produced.
\end{lemma}

\noindent\textbf{Registry contract (verbatim).}
\contractquote{Polar coherence and scalar naturality: for every exact-unit algebra and every two polar data (delta_j, S_j, u_j, h_j), j = 1, 2, for which Pi_{delta_j}: calU x B^{calH}_{delta_j}(J) -> S_j, (U, H) |-> U bold-dot H, is bijective with inverse (u_j, h_j), one has (u_1, h_1) = (u_2, h_2) on S_1 intersect S_2; moreover, for c in U(1) and X, cX in S_j, u_j(cX) = c*u_j(X) and h_j(cX) = h_j(X).}

\paragraph{Proof (prose account of the af-validated tree).}
\emph{Coherence.} Fix $X\in S_1\cap S_2$ and put $p_j=(u_j(X),h_j(X))$, so
$p_j\in\Unit\times B^{\Herm}_{\delta_j}(J)$ and $\Pi_{\delta_j}(p_j)=X$. Let $k$ index the
larger radius; then $B^{\Herm}_{\delta_l}(J)\subseteq B^{\Herm}_{\delta_k}(J)$ for
$l\ne k$, so both $p_1,p_2$ lie in the domain of $\Pi_{\delta_k}$. Every $\Pi_\delta$ has the
\emph{same} formula $(U,H)\mapsto U\bdot H$, so both have $\Pi_{\delta_k}$-image $X$, and the
assumed bijectivity --- hence injectivity --- of $\Pi_{\delta_k}$ gives $p_1=p_2$. That is
\eqref{eq:s1cn-coherence}.

\emph{Scalar naturality.} $\Unit$ is closed under the circle action: if $U\dg\bdot U=J$ and
$U\bdot R=J$ then conjugate-linearity of $\dagger$, bilinearity and $\bar cc=1$ give
$(cU)\dg\bdot(cU)=J$ and $(cU)\bdot(c^{-1}R)=J$, using no associativity and no positivity.
Hence for $p=(u_j(X),h_j(X))$ the pair $p_c=(c\,u_j(X),h_j(X))$ again lies in
$\Unit\times B^{\Herm}_{\delta_j}(J)$, and bilinearity in the first variable gives
$\Pi_{\delta_j}(p_c)=c\,X$. Since $q=(u_j(cX),h_j(cX))$ has the same image, injectivity
gives $q=p_c$, which is \eqref{eq:s1cn-natural}.
