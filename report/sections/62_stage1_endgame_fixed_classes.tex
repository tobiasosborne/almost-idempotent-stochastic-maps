\section{S1-ENDGAME fixed-class rows: the extra fixed class, the projection bridges, the complementary pair, and the fresh two-point inclusion}\label{sec:stage1-endgame-fixed-classes}

This section follows one fixed class from the Stage-1 quotient to the first two-point
operator-space inclusion.  The first row is the topological keystone; the next three
successively turn its phase-corrected fixed unitary into projections for the rectified and
original products; the last row amplifies the complementary pair uniformly.  The ambient
structures are those of Definitions~\ref{def:stage1-polar-witness-data},
\ref{def:approximate-unitary-space}, and \ref{def:extended-epsilon-cstar-algebra}.

\begin{lemma}[\texttt{lem-stage1-extra-fixed-class}]\label{lem:stage1-extra-fixed-class}
There is one universal Stage-1 polar witness tuple
\[
 W=(C_{\rm rect},C_{\rm ch},C_{\rm pol},C_{\rm grp},C_{\rm path},C_{\rm der},
 e_{\rm rect},\kappa_{\rm ch},\kappa_{\rm pol},\kappa_{\rm der},
 \delta_*,\ecs_*^r,e_{S1},r_{\rm iso})
\]
with the six coefficients at least $1$, $0<e_{\rm rect}\le C_{\rm rect}^{-1}$,
$0<\kappa_{\rm ch},\kappa_{\rm pol},\kappa_{\rm der}\le\tfrac12$, and
\begin{align*}

                         \kappa_{\rm pol}/(4C_{\rm pol})\},\\
 \ecs_*^r&=\min\{\tfrac14,\kappa_{\rm ch}/(4C_{\rm ch}),
 \kappa_{\rm pol}/(4C_{\rm pol}),\kappa_{\rm der}/(8C_{\rm der}),
 C_{\rm grp}^{-1},\delta_*/(12C_{\rm path}C_{\rm grp})\},\\
 e_{S1}&=\min\{e_{\rm rect},\ecs_*^r/C_{\rm rect}\},
 &r_{\rm iso}&=\min\{\delta_*/4,\kappa_{\rm der}/(8C_{\rm der})\}.
\end{align*}
There are also universal $e_{\rm quot}^r,\ecs_B^r,C_{\rm fix},e_{\rm fix}^r,r_{\rm bidx}>0$
with $C_{\rm fix}<\infty$, $\ecs_B^r=\min\{\ecs_*^r,e_{\rm quot}^r\}$,
$0<e_{\rm fix}^r\le\ecs_B^r$, $C_{\rm fix}\ge C_{\rm grp}$, such that every
finite-dimensional exact-unit $\ecs_r$-$C^*$-algebra with
$0\le\ecs_r\le e_{\rm fix}^r$ and $1<\dim_{\mathbb C}\calX<\infty$ admits the following
common data.  There are charts $g_V:B_{2\delta_*}^{\aHerm}(0)\to
B_{2\delta_*}^{\Herm}(0)$ for $V\in\Unitc{\delta_*}$, maps
$u_\delta,h_\delta,\mu,\sigma,H$, the quotient $\breve{\Unit}=\mathcal U_e/U(1)$ with
$\breve\mu,\breve\sigma$, a $\breve\sigma$-fixed class $\breve U$, and
$U_0,U\in\mathcal U_e$, $c,a\in U(1)$, satisfying:
\begin{enumerate}
\item[(A$_2$)] $g_V(A^\parallel)$ is the unique point of $B_{2\delta_*}^{\Herm}(0)$ with
$f_V(A^\parallel+g_V(A^\parallel))=0$, where
\[

the $C^1$ charts cover $\Unit$, and
$\nrm{Dg_V(A^\parallel)}\le C_{\rm ch}(\ecs_r+\delta_*)$ and
$\nrm{D_{A^\perp}f_V(A^\parallel+g_V(A^\parallel))-I_{\Herm}}
 \le C_{\rm ch}(\ecs_r+\delta_*)<1$.
\item[(A$_4$)] $\Pi_{\delta_*}(V,K)=V\bdot K$ is a $C^1$ diffeomorphism from
$\Unit\times B_{\delta_*}^{\Herm}(J)$ onto its image $S_{\delta_*}$, with unique inverse

$(u_\delta(V),h_\delta(V))=(V,J)$ for $V\in\Unit$.

\end{align*}
\item[(A$_6$)] If $\nrm{V_1-V_0}\le q\le1$, $C_{\rm path}q\le\tfrac14$, and

$H(t,V_0,V_1)=u_\delta((1-t)V_0+tV_1)$ is a jointly continuous path from $V_0$ to $V_1$
and $H(t,c_0V_0,c_0V_1)=c_0H(t,V_0,V_1)$ for $c_0\in U(1)$.
\item[(A$_7$)] For $s=\pm1$, the chart
$\chi_s(A)=sJ\bdot(J+A+g_{sJ}(A))$ on $B_{r_{\rm iso}}^{\aHerm}(0)$ is preserved by

$\nrm{D(F_s-\mathrm{id})(A)+2I}\le C_{\rm der}(\ecs_r+r_{\rm iso})$.
\item[(R)] For $q_*=C_{\rm grp}\ecs_r$,
$r_-=\delta_*-C_{\rm pol}(\ecs_r\delta_*+\delta_*^2)$ and
$\eta_*=C_{\rm path}(q_*+\ecs_rq_*+q_*^2)$, all the displayed graph, polar, path, and
derivative guards hold: in particular $q_*<r_-$, $C_{\rm path}q_*\le\tfrac14$,
$C_{\rm ch}(\ecs_r+\delta_*)\le\kappa_{\rm ch}$,
$C_{\rm pol}(\ecs_r+\delta_*)\le\kappa_{\rm pol}$, $\eta_*<r_-$,
$C_{\rm der}(\ecs_r+r_{\rm iso})\le\kappa_{\rm der}/4<1$, and
$(1+\ecs_r)(1+C_{\rm ch}(\ecs_r+\delta_*))r_{\rm iso}+q_*<2\delta_*$.
\end{enumerate}
Moreover $r_{\rm bidx}=r_{\rm iso}$, $\breve e=[J]$,
$\breve\mu([V_1],[V_2])=[\mu(V_1,V_2)]$, $\breve\sigma([V])=[\sigma(V)]$, and
$(\breve\Unit,\breve\mu,\breve e)$ is a connected H-space with smooth left inversion
$\breve\sigma$.  The quotient is a connected compact orientable smooth manifold without
boundary of real dimension $\dim_{\mathbb C}\calX-1$, homeomorphic to a finite simplicial
complex; $\breve e$ is an isolated fixed point of local index $+1$; and $J,-J$ are the only
$\sigma$-fixed points in their respective ambient $r_{\rm bidx}$-balls.  Finally
\[
 \breve U\ne\breve e,\quad [U_0]=\breve U,\quad \sigma(U_0)=cU_0,\quad a^2=c,\quad
 U=aU_0,\quad [U]=\breve U,\quad \sigma(U)=U,
\]
$\sigma(c_0V)=\overline{c_0}\sigma(V)$ and
$\nrm{\sigma(V)-V\dg}\le C_{\rm grp}\ecs_r$, while
$\nrm{U-U\dg}\le C_{\rm fix}\ecs_r$ and
$\nrm{U-J},\nrm{U+J}\ge r_{\rm bidx}$.
\end{lemma}

\paragraph{Proof (prose account of the af-validated tree).}
The tree first selects the tuple $W$ from
Lemma~\ref{lem:stage1-polar-constant-ledger} exactly once.  For this same tuple,
Lemma~\ref{lem:stage1-bound-quotient-index-data} supplies the conditional quotient package.
The scalar branch then derives all its guards directly from the minimum formulas:
$C_{\rm ch}(\ecs_r+\delta_*),C_{\rm pol}(\ecs_r+\delta_*)$ are at most half their
respective margins, $r_-\ge3\delta_*/4$, $q_*\le\delta_*/12<r_-$, and
$\eta_*\le61\delta_*/576<r_-$.  It also obtains the derivative bound and
$(1+\ecs_r)(1+C_{\rm ch}(\ecs_r+\delta_*))r_{\rm iso}+q_*<2\delta_*$.  Thus one may take
$e_{\rm fix}^r=\ecs_B^r$ and $C_{\rm fix}=C_{\rm grp}$ and instantiate the analytic and
topological data on common objects.

Suppose now that $\breve e$ were the only fixed class.  The finite simplicial model and
Lemma~\ref{lem:finite-polyhedron-maximal-simplex-placement} put that point in a maximal
simplex.  Lemmas~\ref{lem:topology-lefschetz-hopf} and
\ref{lem:topology-local-index-sign} then give $L(\breve\sigma)=1$.  On the other hand,
Lemma~\ref{lem:stage1-left-inversion-trace} gives
$\operatorname{Tr}(\breve\sigma^{*k})=(-1)^k b_k$, so Definition~\ref{def:lefschetz-fixed-point-data}
yields $L(\breve\sigma)=\sum_k b_k$.  Connectedness gives $b_0=1$, while orientability
and positive dimension give $b_d\ge1$ by Lemma~\ref{lem:topology-orientable-top-cohomology};
hence $L(\breve\sigma)\ge2$, a contradiction.

For an extra fixed class $\breve U$, the phase-lift clause returns
$[U_0]=\breve U$, $\sigma(U_0)=cU_0$ and $a^2=c$.  Scalar covariance makes
$U=aU_0$ fixed and gives $\nrm{U-U\dg}\le C_{\rm fix}\ecs_r$.  The two local isolation
statements force $\nrm{U-J},\nrm{U+J}\ge r_{\rm bidx}$, since either strict inequality
would put $U$ in the scalar class.  Three recorded challenges were repaired: the bound
package was restricted to its actual scope, the Lefschetz step was phrased on the same
finite-polyhedron presentation rather than through an unstated conjugacy-invariance
claim, and a sibling dependency was made explicit.

\begin{lemma}[\texttt{lem-stage1-fixed-unitary-projection-bridge}]\label{lem:stage1-fixed-unitary-projection-bridge}
There are universal $C_{\rm bridge}<\infty$ and $e_{\rm bridge}^r>0$ such that every
finite-dimensional exact-unit $\ecs_r$-$C^*$-algebra with
$0\le\ecs_r\le e_{\rm bridge}^r$ and $1<\dim_{\mathbb C}\calX<\infty$ contains a
nontrivial $C_{\rm bridge}\ecs_r$-projection for the product $\bdot$ and unit $J$.
\end{lemma}

\noindent\textbf{Registry contract (verbatim).} \contractquote{Fixed-unitary projection bridge: there are universal C_bridge<infinity and e_bridge^r>0 such that every finite-dimensional exact-unit epsilon_r-C*-algebra with 0<=epsilon_r<=e_bridge^r and 1<dim_C calX<infinity contains a nontrivial C_bridge*epsilon_r-projection P for the product bold-dot and unit J.}

\paragraph{Proof (prose account of the af-validated tree).}
Apply Lemma~\ref{lem:stage1-extra-fixed-class} once, obtaining $C_{\rm fix}$,
$e_{\rm fix}^r$, $r=r_{\rm bidx}$, and $U$ with
$\nrm{U-U\dg}\le C_{\rm fix}\ecs_r$ and $\nrm{U\pm J}\ge r$.  The tree fixes
\[
 K=3+4C_{\rm fix}/r,\quad C_{\rm bridge}=\max\{C_{\rm fix},K\},\quad
 e_{\rm bridge}^r=\min\{e_{\rm fix}^r,\tfrac12,C_{\rm fix}^{-1},r/C_{\rm fix}\}.
\]
Set $A=(U+U\dg)/2$, $P=(J+A)/2$, and $Q=J-P$.  The unitary identity
$U\dg\bdot U=J$ and the lower $C^*$-bound give $\nrm U<2$, while
$P\bdot P-P=(A\bdot A-J)/4$ and $Q\bdot Q-Q=P\bdot P-P$.  Expanding around
$U\dg\bdot U$ gives $\nrm{A\bdot A-J}\le3\nrm{U-U\dg}$, hence both defects are
at most $C_{\rm bridge}\ecs_r$.

The exact identities $U+J=2P+(U-A)$ and $U-J=-2Q+(U-A)$, together with
$\nrm{U-A}\le r/2$, imply $\nrm P,\nrm Q\ge r/4$.  If $R$ is either $P$ or $Q$, the
upper and lower $C^*$ estimates applied to $R\bdot R$ bound $\nrm R<3$ and then
$|\nrm R-1|\le(3+4C_{\rm fix}/r)\ecs_r$.  Thus Definition~\ref{def:delta-projection}
recognizes $P$ as nontrivial.  The endpoint $\ecs_r=0$ uses only non-strict bounds.
Two verifier challenges found an unbound $Q$ in leaf statements; both were repaired by
defining $Q=J-P$ locally before the affected estimates.

\begin{lemma}[\texttt{lem-stage1-rectified-nontrivial-projection}]\label{lem:stage1-rectified-nontrivial-projection}
There are universal $C_{\rm proj}<\infty$ and $e_{\rm proj}>0$ such that every
finite-dimensional extended $\ecs_X$-$C^*$-algebra $(\calX,I_X,\cdot,\dagger)$ with
$0\le\ecs_X\le e_{\rm proj}$ and $1<\dim_{\mathbb C}\calX<\infty$ contains a
nontrivial $C_{\rm proj}\ecs_X$-projection $P_0$ for the original product and unit $I_X$.
\end{lemma}

\noindent\textbf{Registry contract (verbatim).} \contractquote{There are universal C_proj<infinity and e_proj>0 such that every finite-dimensional extended epsilon_X-C*-algebra (calX,I_X,.,dagger) with 0<=epsilon_X<=e_proj and 1<dim_C calX<infinity contains a nontrivial C_proj*epsilon_X-projection P_0 for the original product and original unit I_X.}

\paragraph{Proof (prose account of the af-validated tree).}
Choose the rectification constants $C_{\rm rect},e_{\rm rect}$ and the bridge constants
$C_{\rm bridge},e_{\rm bridge}^r$.  With $D=\max\{1,C_{\rm bridge}\}$, the tree fixes a
universal complement coefficient $K$, then
$B=K(D+1)C_{\rm rect}$, $A=C_{\rm rect}(D+4)$,
$C_{\rm proj}=A+6$, and
$e_{\rm proj}=\min\{e_{\rm rect},e_{\rm bridge}^r/C_{\rm rect},1,B^{-1}\}$.
Lemma~\ref{lem:stage1-rectified-cstar-control} puts an exact-unit product $\bdot$ and unit
$J$ on the same involutive normed space, with $\ecs_r=C_{\rm rect}\ecs_X$,
$\nrm{J-I_X}\le C_{\rm rect}\ecs_X$, and product discrepancy at most
$C_{\rm rect}\ecs_X\nrm x\nrm y$.  Lemma~\ref{lem:stage1-fixed-unitary-projection-bridge}
then supplies a nontrivial rectified projection $P$.

Its nonvanishing estimates give $|\nrm P-1|,|\nrm{J-P}-1|\le B\ecs_X$ and hence
$\nrm P\le2$.  Product comparison transfers the defect:
$\nrm{P^2-P}\le C_{\rm rect}\ecs_X\nrm P^2+C_{\rm bridge}C_{\rm rect}\ecs_X
\le A\ecs_X$.  For $R=I_X-P$, the original approximate-unit errors contribute at most
$6\ecs_X$, so $\nrm{R^2-R}\le C_{\rm proj}\ecs_X$.  Finally unit closeness transfers the
nonvanishing estimate from $J-P$ to $I_X-P$.  Thus $P_0=P$ is nontrivial in the sense of
Definition~\ref{def:delta-projection}.  The seven-node tree validated first-pass with no
substantive challenge; its only transient blocked verdict cleared when dependencies did.

\end{lemma}

$\overline C=\max\{C_{\rm proj},C_{\rm proj}+C_{\rm cmp}\}$; both $P'$ and $P''$ are then
nonvanishing $\overline C\ecs_X$-projections.  Export node 1.4 asserts the exact identities $P_0I_X=I_XP_0=P_0$ and hence the exact expansion $P_0-P_0^2$; however, the registry amendment records that this exact-unit move was rejected and that the cross-products equal $P_0-P_0^2$ only up to controlled unit error.  These two canonical records are inconsistent, so this row is not a reconciled proof account.  Taking $C_{\rm np}=\overline C$ and

automatically an exact two-sided unit.  The repaired live tree factors that leaf into the
general-unit complement-defect clause and a separate nonvanishing clause, and enlarges

\begin{lemma}[\texttt{lem-stage1-fresh-two-point-inclusion}]\label{lem:stage1-fresh-two-point-inclusion}
There are universal $C_{\rm pair}<\infty$ and $e_{\rm pair}>0$ such that every
finite-dimensional extended $\ecs_X$-$C^*$-algebra $(\calX,I_X,\cdot,\dagger)$ with
$0\le\ecs_X\le e_{\rm pair}$ and $1<\dim_{\mathbb C}\calX<\infty$ contains nonvanishing
$C_{\rm pair}\ecs_X$-projections $P',P''$ with $P'+P''=I_X$ for which
\[
 v^{(2)}:\mathbb C^2\longrightarrow\calX,\qquad
 v^{(2)}(\lambda,\mu)=\lambda P'+\mu P'',
\]
is an extended $C_{\rm pair}\ecs_X$-inclusion, satisfies $v^{(2)}(1,1)=I_X$, and sends
the standard projection basis $\Pi',\Pi''$ to $P',P''$, respectively.
\end{lemma}

\paragraph{Proof (prose account of the af-validated tree).}

\[
 d=\max\{\nrm{q_1^2-q_1},\nrm{q_2^2-q_2},\nrm{q_1q_2},\nrm{q_2q_1}\}
 \le C_{\rm np}\ecs_X.
\]
After one universal shrink, nonvanishing gives $\nrm{q_i}\ge\tfrac12$.  Define the single
level-one map $v^{(2)}(\lambda,\mu)=\lambda q_1+\mu q_2$ and only its canonical
amplifications $v_n(A,B)=A\otimes q_1+B\otimes q_2$.  Definition~\ref{def:operator-space}
gives $\nrm{T\otimes z}_n=\nrm T\nrm z$: the upper bound is the rectangular matrix
inequality and the reverse bound comes from compression by norming unit vectors.

Bilinearity now yields the exact four-term defect identity
\[
 v_n(x)v_n(y)-v_n(xy)=AC\otimes(q_1^2-q_1)+AD\otimes q_1q_2
 +BC\otimes q_2q_1+BD\otimes(q_2^2-q_2),
\]
so every amplification has multiplicative defect at most $4d$.  If
$\max\{\nrm A,\nrm B\}=1$, multiplication by $I_n\otimes q_1$ or
$I_n\otimes q_2$ gives
$\nrm{v_n(A,B)}\ge(1-4d)/(1+\ecs_X)$; after shrinking, this is at least $1/4$.
