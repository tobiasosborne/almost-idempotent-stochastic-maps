\section{Residual distance bound}\label{sec:residual-upper}

\begin{lemma}[Registry contract for \texttt{lem-residual-upper}]
\label{lem:residual-upper}
\contractquote{Residual distance bound: let C be the convex hull of finitely many points of R^n, let b_1..b_M, c_1..c_N >= 0 with m = sum_j b_j - sum_k c_k > 0, let p_j, r_k be points of R^n with q = (sum_j b_j p_j - sum_k c_k r_k) / m, and let D_k >= 0 satisfy ||x - r_k||_1 <= D_k for all x in C and each k; then m * dist_1(q, C) <= sum_j b_j * dist_1(p_j, C) + sum_k c_k * D_k.}
\end{lemma}
